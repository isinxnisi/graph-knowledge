export function rememberPreviewEntry(){

 if(!href.startsWith('/communities'))sessionStorage.setItem('preview-entry',href);
}

export function PreviewReturn(){const {locale}=useUiLocale();const [href,setHref]=useState<string|null>(null);useEffect(()=>{const v=sessionStorage.getItem('preview-entry');if(document.referrer.startsWith(location.origin+'/')&&v&&/^\/(?:member|explore\/|en|$)/.test(v))setHref(v);},[]);return href?<a className="preview-return" href={href}>{locale==='en'?'← Return to where you left off':'← 元の画面へ戻る'}</a>:null;}
export function CommunityFrame(){const {locale}=useUiLocale();const [src,setSrc]=useState('');useEffect(()=>{sessionStorage.setItem('community-connection-lang',locale);setSrc('/study/index.html'+(location.hash||'#'+encodeURIComponent(JSON.stringify({page:'directory'}))));const receive=(e:MessageEvent)=>{const frame=document.querySelector<HTMLIFrameElement>('#community-frame');if(e.origin!==location.origin||e.source!==frame?.contentWindow)return;const d=e.data;if(d?.kind==='route'&&typeof d.hash==='string'&&d.hash.length<1500){history.replaceState(history.state,'','/communities'+d.hash);}if(d?.kind==='height'&&Number.isFinite(d.height)){if(frame){frame.style.height=Math.min(Math.max(d.height,500),20000)+'px';if(!frame.dataset.restored){frame.dataset.restored='1';requestAnimationFrame(()=>scrollTo(0,Number(sessionStorage.getItem('preview-community-scroll')||0)));}}}if(d?.kind==='graph'&&['c1','c2','c3'].includes(d.capsule)){sessionStorage.setItem('preview-community-return',location.pathname+location.hash);sessionStorage.setItem('preview-community-scroll',String(scrollY));const refs:Record<string,string>={c1:'2b7c6289-d050-4dc0-93e3-9307681253cc',c2:'0820c3e8-7219-4e57-abb0-7d04e253fd4a',c3:'2b7c6289-d050-4dc0-93e3-9307681253cc'};const target=(locale==='en'?'/en':'')+'/explore/'+refs[d.capsule];sessionStorage.setItem('preview-community-target',target);location.assign(target);}if(d?.kind==='account')location.assign('/member#community-membership');if(d?.kind==='home')location.assign(locale==='en'?'/en':'/');};addEventListener('message',receive);return()=>removeEventListener('message',receive);},[locale]);return <>{src&&<iframe id="community-frame" title={locale==='en'?'Community study':'コミュニティ試作'} src={src} className="community-frame"/>}</>}

export function ListingLink({id}:{id:string}){const {locale}=useUiLocale();return <a data-testid="listing-link" href={'/communities#'+encodeURIComponent(JSON.stringify({page:'community',cid:id.endsWith('3')?'studio':'atelier',tab:'capsules'}))} onClick={rememberPreviewEntry}>{locale==='en'?'Listed in a community →':'コミュニティの掲載先 →'}</a>}
