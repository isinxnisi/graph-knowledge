import {NextRequest,NextResponse} from 'next/server';
// The POST allowlist contains existing read-only discovery queries, never writes.
const read=/^\/api\/(?:dataset|entities(?:\/[0-9a-f-]{36}(?:\/(?:overview|neighbors))?)?|claims\/[0-9a-f-]{36}|discovery\/(?:entities|subject-matches|paths|witness|claims))$/;
const post=/^\/api\/discovery\/(?:entities|subject-matches|paths|witness)$/;
async function handle(req:NextRequest){
 const path=req.nextUrl.pathname;
 if(path.startsWith('/api/member/')||path.startsWith('/api/auth/'))return NextResponse.json({error:{code:'LAB_ACCOUNT_NOT_CONNECTED'}},{status:503});
 if(!read.test(path)||(req.method!=='GET'&&!(req.method==='POST'&&post.test(path))))return NextResponse.json({error:{code:'LAB_NOT_CONNECTED'}},{status:404});
 if(req.method==='POST'&&Number(req.headers.get('content-length')||0)>100000)return new NextResponse(null,{status:413});
 const body=req.method==='POST'?await req.text():undefined;
 if(body&&new TextEncoder().encode(body).length>100000)return new NextResponse(null,{status:413});
 try{
  const r=await fetch('http://127.0.0.1:3200'+path+req.nextUrl.search,{method:req.method,body,headers:{'Content-Type':'application/json'},cache:'no-store',redirect:'error',signal:AbortSignal.timeout(20000)});
  return new NextResponse(await r.text(),{status:r.status,headers:{'Content-Type':'application/json','Cache-Control':'no-store'}});
 }catch{return NextResponse.json({error:{code:'LAB_KG_UNAVAILABLE'}},{status:503});}
}
export const GET=handle;
export const POST=handle;
