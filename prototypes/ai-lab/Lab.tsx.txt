function title(id:string,en:boolean){return titles[id]?.[en?1:0]||id;}
function when(value:string,en:boolean){return new Date(value).toLocaleString(en?'en-GB':'ja-JP',{timeZone:'Asia/Tokyo',hour12:false})+' JST';}
function kind(run:Run,en:boolean){
 if(run.executionKind==='rule')return en?'Non-AI baseline':'非AIの比較基準';
 if(run.executionKind==='kev-api')return run.status==='completed'?(en?'Real AI inference':'AI実推論'):(en?'Real AI API attempt':'AI実API試行');
 if(run.executionKind==='interrupted')return en?'Interrupted; dispatch unknown':'中断・送信状況不明';
 return en?'Not connected':'未接続';

function status(run:Run,en:boolean){return ({completed:en?'Completed':'完了',held:en?'Held before inference':'推論前に保留',failed:en?'Failed':'失敗',unknown:en?'Outcome unknown':'結果不明'} as Record<string,string>)[run.status]||run.status;}

  <p>{en?'Original Claim':'元のClaim'}: {names[step.subjectId]?.name} → {names[step.objectId]?.name} · {step.predicate} · {step.polarity}</p>
  <p>{en?'Traversal':'辿り方向'}: {step.direction} · {scope?`${scope.volumeLabel} / ISBN ${scope.isbn13}`:(en?'Publication scope unspecified':'刊行物scope未指定')}</p>
  <a href={(en?'/en':'/')+'?read=claim&dataset=public-kg&id='+step.claimId} onClick={()=>rememberDeparture((en?'/en':'/')+'?read=claim&dataset=public-kg&id='+step.claimId)}>{en?'Read Claim and evidence →':'Claimと根拠を読む →'}</a>

 </li>;})}</ol>;
}
