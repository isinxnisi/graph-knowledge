 return <img className="apprentice-avatar" data-ai-id={aiId} data-artwork={selected} src={'/apprentice-study/'+artwork[selected]+'.png'} width={size} height={size} alt={decorative?'':en?'Apprentice exploration AI · B′ study':'探索者見習いAI・B′仮デザイン'} draggable={false}/>;
}
