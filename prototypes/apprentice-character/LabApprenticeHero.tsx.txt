"use client";
import {useRef,type ReactNode} from 'react';
import {ApprenticeAvatar} from './Apprentice';
import {useMotionEnvironment,useMotionOnScreen,type MotionStrength} from './motion';

// Presentation only. Change this internal parameter to compare strong motion;
// it is never derived from activity records, URL parameters or model state.
const LAB_AVATAR_STRENGTH:MotionStrength='standard';

export default function LabApprenticeHero({aiId,en,children}:{aiId:string;en:boolean;children:ReactNode}){
 const {reduced,visible}=useMotionEnvironment(),ref=useRef<HTMLDivElement>(null),onScreen=useMotionOnScreen(ref);
 return <header className="lab-hero lab-apprentice-focus" id="apprentice" data-testid="explorer-apprentice" data-ai-id={aiId}>
  <div ref={ref} className="motion-stage lab-apprentice-motion" data-testid="lab-motion-avatar" data-state="idle" data-strength={LAB_AVATAR_STRENGTH} data-quiet={false} data-moving={!reduced&&visible&&onScreen}>
   <div className="motion-body motion-png"><ApprenticeAvatar aiId={aiId} size={128} en={en} decorative/></div>
   <span className="lab-avatar-motion-note">{en?'Floating artwork':'表示上の浮遊演出'}</span>
  </div>
  {children}
 </header>;
}
