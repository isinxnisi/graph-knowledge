/** Opt-in preview on the copied real pages; never a posted community message or an AI run. */
export function CharacterContext(){
 const pathname=usePathname();const [selected,setSelected]=useState<Candidate|null>(null);const [expanded,setExpanded]=useState(true);
 useEffect(()=>{
  if(!pathname||!/^\/(?:en\/?|(?:en\/)?explore\/[0-9a-f-]{36}|member)?$/.test(pathname)){setSelected(null);return;}
  const query=candidate(new URLSearchParams(location.search).get('character-study'));
  if(query){remember(query.id);setSelected(query);return;}
  try{setSelected(candidate(sessionStorage.getItem(storageKey))||null);}catch{setSelected(null);}
 },[pathname]);
 if(!selected)return null;
 function stop(){try{sessionStorage.removeItem(storageKey);}catch{/* Clear current view regardless. */}const url=new URL(location.href);url.searchParams.delete('character-study');history.replaceState(null,'',url);setSelected(null);}
 function choose(id:string){const next=candidate(id);if(!next)return;remember(id);const url=new URL(location.href);url.searchParams.set('character-study',id);history.replaceState(null,'',url);setSelected(next);}
 return <aside className="character-context" data-testid="character-context" aria-label="公式AIの配置比較">
  <div className="character-context-bar"><strong>公式AI · 配置比較</strong><label>案 <select aria-label="比較する案" value={selected.id} onChange={e=>choose(e.target.value)}>{candidates.map(item=><option value={item.id} key={item.id}>{item.letter} · {item.name}</option>)}</select></label><a href="/ai-character-study">比較一覧へ</a><button type="button" onClick={()=>setExpanded(!expanded)} aria-expanded={expanded}>{expanded?'サンプルを畳む':'サンプルを表示'}</button><button type="button" onClick={stop}>比較を終了</button></div>
  {expanded&&<div className="character-context-sample" data-testid="character-sample"><p className="character-sample-label">配置サンプル · 投稿・応答ではありません</p><div className="character-guide"><Avatar item={selected} size={64}/><div><strong>Plunode公式AI</strong><p>案内と交流、探索AIの育成を担当。</p></div></div><div className="character-comment"><Avatar item={selected} size={20}/><span><b>Plunode公式AI</b><small>20px · コメント横の表示例</small></span></div><p className="character-context-note">探索者見習いAIとは別個体。<br/>正式採用・アニメーションは未決定です。</p></div>}
 </aside>;
}
