import {officialAi,mentoringDemos} from './official-ai';

// Explicit supported sizes avoid ambiguous styling between the three approved ranges.
export type OfficialAiSize=16|18|20|24|48|64|80|96|128;
type AvatarProps={en?:boolean;decorative?:boolean}&(
 {usage:'profile';size?:128}|{usage:'regular';size?:48|64|80|96}|{usage:'comment';size?:16|18|20|24}
);
export function OfficialAiAvatar({usage,size,en=false,decorative=false}:AvatarProps){
 const pixels=size??(usage==='profile'?128:usage==='regular'?64:20);
 const variant=pixels<=24?'symbolic-flat':pixels<=96?'soft-flat':'translucent';
 return <img className="official-ai-avatar" data-identity-ref={officialAi.identityRef.key} data-avatar-usage={usage} data-avatar-variant={variant}
  src={'/character-study/v02-'+variant+'.png'} width={pixels} height={pixels}
  alt={decorative?'':officialAi.label[en?'en':'ja']} draggable={false}/>;
}

export function LabMentorship({en}:{en:boolean}){
 return <section className="lab-mentorship" id="mentorship" aria-label={en?'Mentoring comments':'育成コメント'}>
  <section className="lab-mentor-comments" id="mentoring-comments" aria-labelledby="mentoring-title">
   <header className="lab-mentor-comments-head"><OfficialAiAvatar usage="regular" size={48} en={en} decorative/><div><p className="lab-eyebrow">{officialAi.label[en?'en':'ja']} · {en?'mentor':'育成担当'}</p><h2 id="mentoring-title">{en?'Mentoring comments':'育成コメント'} <span className="lab-demo-label">DEMO</span></h2></div></header>
   <a className="lab-mentor-about" href="/ai-character-study#official-profile">{en?'About the mentor and its community role →':'育成担当のプロフィール・コミュニティでの役割 →'}</a>
   <p className="lab-demo-disclosure">{en?'Display examples only. These are not comments generated or posted by the AI, nor explanations taken from experiment records.':'画面確認用の例文です。AIが生成・投稿した発言でも、実験記録の解説でもありません。'}</p>
   <ul className="lab-mentor-comment-list">{mentoringDemos.map(comment=><li className="lab-mentor-comment" key={comment.id} data-comment-kind={comment.kind} data-display-identity-ref={comment.displayIdentityKey}>
    <OfficialAiAvatar usage="comment" en={en} decorative/>
    <div className="lab-mentor-comment-copy"><div className="lab-mentor-comment-byline"><strong>{en?'Official AI · display example':'公式AI・表示例'}</strong><span className="lab-demo-label">DEMO</span></div><p>{comment[en?'en':'ja']}</p></div>
   </li>)}</ul>
   <p className="lab-mentor-comments-foot">{en?'No real comments yet. These examples are excluded from activity history and growth records.':'実際のコメントはまだありません。この例文は活動履歴・成長記録に含めません。'}</p>
  </section>
 </section>;
}
