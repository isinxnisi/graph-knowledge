import {ApprenticeAvatar} from './Apprentice';
import {officialAi,mentoringDemos} from './official-ai';

// Explicit supported sizes avoid ambiguous styling between the three approved ranges.
export type OfficialAiSize=16|18|20|24|48|64|80|96|128;
type AvatarProps={en?:boolean;decorative?:boolean}&(
 {usage:'profile';size?:128}|{usage:'regular';size?:48|64|80|96}|{usage:'comment';size?:16|18|20|24}
);
export function OfficialAiAvatar({usage,size,en=false,decorative=false}:AvatarProps){
 const pixels=size??(usage==='profile'?128:usage==='regular'?64:20);
 const variant=pixels<=24?'symbolic-flat':pixels<=96?'soft-flat':'translucent';
 return <img className="official-ai-avatar" data-identity-ref={officialAi.identityRef.key} data-avatar-usage={usage} data-avatar-variant={variant}
  src={'/character-study/v02-'+variant+'.png'} width={pixels} height={pixels}
  alt={decorative?'':officialAi.label[en?'en':'ja']} draggable={false}/>;
}
